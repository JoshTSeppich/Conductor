lflags: icanon isig iexten echo echoe -echok echoke -echonl echoctl
	-echoprt -altwerase -noflsh -tostop -flusho -pendin -nokerninfo
	-extproc

	-ignbrk brkint -inpck -ignpar -parmrk
oflags: opost onlcr -oxtabs -onocr -onlret

	-dtrflow -mdmbuf

	eol2 = <undef>; erase = ^?; intr = ^C; kill = ^U; lnext = ^V;

	stop = ^S; susp = ^Z; time = 0; werase = ^W;

=== SIGWINCH ===
(static measurement; SIGWINCH delivery requires an attached client and window resize)
tmux allocates PTY for all panes regardless of spawn context; SIGWINCH is sent
by tmux to the pane process on client resize. Not dependent on spawning env.
